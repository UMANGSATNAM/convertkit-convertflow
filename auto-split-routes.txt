  landingJson.order = [];
  
  for (const sec of sectionsConfig) {

      const content = fs.readFileSync(path.join(SECTIONS_DIR, `${sec.filename}.liquid`), 'utf-8');
      filesToUpsert.push({ filename: `sections/${sec.filename}.liquid`, body: { type: 'TEXT', value: content } });

        settings: {}
      };
